CLASS ZCL_PY_PROVIDER_APP IMPLEMENTATION.


* <SIGNATURE>---------------------------------------------------------------------------------------+
* | Instance Public Method ZCL_PY_PROVIDER_APP->ADD_EMPLOYEE_DATA
* +-------------------------------------------------------------------------------------------------+
* | [--->] PERNR                          TYPE        PERNR-PERNR
* | [--->] P0001S                         TYPE        TT_P0001
* | [--->] P0002                          TYPE        P0002
* | [--->] P0007S                         TYPE        TT_P0007
* | [--->] P0008S                         TYPE        TT_P0008
* | [--->] P0032                          TYPE        P0032
* | [--->] P0130S                         TYPE        TT_P0130
* | [--->] P2006                          TYPE        P2006
* +--------------------------------------------------------------------------------------</SIGNATURE>
  METHOD add_employee_data.

    DATA b2_cluster           TYPE hrf_tim_b2.
    DATA periods              TYPE tt_periods.
    DATA first_period         TYPE ts_period.
    DATA last_period          TYPE ts_period.
    DATA period               TYPE ts_period.
    DATA p0001                TYPE p0001.
    DATA p0008                TYPE p0008.
    DATA p0007                TYPE p0007.
    DATA p0130                TYPE p0130.
    DATA period_valid         TYPE abap_bool.
    DATA p0007_valid          TYPE abap_bool.
    DATA p0008_valid          TYPE abap_bool.
    DATA begda                TYPE begda.
    DATA endda                TYPE endda.
    DATA evaluated_wage_types TYPE tt_pbwla.
    DATA hourly_rate          TYPE pbwla-betrg.
    DATA overtime_number      TYPE ptr_f_zl-anzhl.
    DATA hours_reset_number   TYPE ptr_f_zl-anzhl.
    DATA forced_payment_date  TYPE ptr_f_zl-datum.
    DATA message_handler      TYPE REF TO cl_hrpa_message_list.
    DATA row                  TYPE REF TO data.

    FIELD-SYMBOLS <colors> TYPE lvc_t_scol.

    IF me->params-payroll_begda IS NOT INITIAL AND
       me->params-payroll_endda IS NOT INITIAL.
      b2_cluster = me->get_b2_cluster( pernr ).

      " Do not calculate periods on B2 ZL '9EOP - Tech:End of time period' wage type
      " periods = me->build_periods( b2_cluster ).

      " Period is built based on the T511K table.
      INSERT VALUE #( begda = me->params-payroll_begda
                      endda = me->params-payroll_endda
                      end   = me->params-payroll_endda ) INTO TABLE periods.
    ELSE.
      INSERT VALUE #( begda = me->params-begda
                      endda = me->params-endda
                      end   = me->params-endda ) INTO TABLE periods.
    ENDIF.

    first_period = VALUE #( periods[ 1 ] ).
    last_period  = VALUE #( periods[ lines( periods ) ] ).

    PROVIDE FIELDS * FROM periods INTO period VALID period_valid
                                  BOUNDS begda AND endda
            FIELDS * FROM p0007s  INTO p0007 VALID p0007_valid
                                  BOUNDS begda AND endda
            FIELDS * FROM p0008s  INTO p0008 VALID p0008_valid
                                  BOUNDS begda AND endda
           BETWEEN first_period-begda AND last_period-endda.

      begda = COND #( WHEN p0008_valid EQ abap_true THEN p0008-begda
                      WHEN p0007_valid EQ abap_true THEN p0007-begda
                      ELSE period-begda ).

      endda = COND #( WHEN p0008_valid EQ abap_true THEN p0008-endda
                      WHEN p0007_valid EQ abap_true THEN p0007-endda
                      ELSE period-endda ).

      IF me->params-data_for_end_of_period EQ abap_true AND endda NE period-end.
        CONTINUE.
      ENDIF.

      p0001 = me->get_last_p0001( p0001s = p0001s
                                  begda  = period-begda
                                  endda  = period-endda ).

      p0130 = me->get_last_p0130( p0130s = p0130s
                                  begda  = period-begda
                                  endda  = period-endda ).

      row = me->create_row( ).

      IF me->params-data_for_end_of_period EQ abap_false AND endda NE period-end.
        ASSIGN row->('COLOR') TO <colors>.
        IF sy-subrc EQ 0.
          " Add row color if data does not overlap end of period!
          INSERT VALUE #( color = VALUE #( col = 3 int = 0 inv = 1 ) ) INTO TABLE <colors>.
        ENDIF.
      ENDIF.

      ASSIGN row->('END_OF_PERIOD') TO FIELD-SYMBOL(<end_of_period>).
      IF sy-subrc EQ 0.
        <end_of_period> = period-end.
      ENDIF.

      ASSIGN row->('IS_IT0130_LOCKED') TO FIELD-SYMBOL(<is_it0130_locked>).
      IF sy-subrc EQ 0.
        <is_it0130_locked> = COND abap_bool( WHEN p0130-reldt >= period-endda THEN abap_true
                                                                              ELSE abap_false ).

        ASSIGN row->('IT0130_LOCKED') TO FIELD-SYMBOL(<it0130_locked>).
        IF sy-subrc EQ 0.
          <it0130_locked> = SWITCH string( <is_it0130_locked> WHEN abap_true THEN me->icon( icon = icon_locked
                                                                                            info = |Locked until { p0130-reldt DATE = ENVIRONMENT }| )
                                                                             ELSE me->icon( icon_unlocked ) ).
        ENDIF.
      ENDIF.

      ASSIGN row->('PERNR') TO FIELD-SYMBOL(<pernr>).
      IF sy-subrc EQ 0.
        <pernr> = pernr.
      ENDIF.

      ASSIGN row->('SNAME') TO FIELD-SYMBOL(<sname>).
      IF sy-subrc EQ 0.
        <sname> = |{ p0002-vorna } { p0002-nachn }|.
      ENDIF.

      ASSIGN row->('PNALT') TO FIELD-SYMBOL(<pnalt>).
      IF sy-subrc EQ 0.
        <pnalt> = p0032-pnalt.
      ENDIF.

      ASSIGN row->('WERKS') TO FIELD-SYMBOL(<werks>).
      IF sy-subrc EQ 0.
        <werks> = p0001-werks.
      ENDIF.

      ASSIGN row->('POSITION') TO FIELD-SYMBOL(<position>).
      IF sy-subrc EQ 0.
        <position> = me->get_position_text( id    = p0001-plans
                                            begda = p0001-endda
                                            endda = p0001-endda ).
      ENDIF.

      ASSIGN row->('BEGDA') TO FIELD-SYMBOL(<begda>).
      IF sy-subrc EQ 0.
        <begda> = begda.
      ENDIF.

      ASSIGN row->('ENDDA') TO FIELD-SYMBOL(<endda>).
      IF sy-subrc EQ 0.
        <endda> = endda.
      ENDIF.

      evaluated_wage_types = me->get_evaluated_wage_types( pernr  = pernr
                                                           begda  = p0008-begda
                                                           endda  = p0008-endda
                                                           p0001s = p0001s
                                                           p0007s = p0007s
                                                           p0008s = p0008s ).

      ASSIGN row->('MONTHLY_SALARY') TO FIELD-SYMBOL(<monthly_salary>).
      IF sy-subrc EQ 0.
        <monthly_salary> = me->get_monthly_salary( evaluated_wage_types ).
      ENDIF.

      ASSIGN row->('CURRENCY_SALARY') TO FIELD-SYMBOL(<currency_salary>).
      IF sy-subrc EQ 0.
        <currency_salary> = p0008-waers.
      ENDIF.

      ASSIGN row->('BONUS_PAYOUT') TO FIELD-SYMBOL(<bonus_payout>).
      IF sy-subrc EQ 0.
        DATA(bonus_wt) = VALUE #( evaluated_wage_types[ lgart = me->params-wage_type-bonus_payout ] OPTIONAL ).
        <bonus_payout> = bonus_wt-betrg.
      ENDIF.

      ASSIGN row->('CURRENCY_BONUS') TO FIELD-SYMBOL(<currency_bonus>).
      IF sy-subrc EQ 0.
        <currency_bonus> = p0008-waers.
      ENDIF.

      ASSIGN row->('WEEKLY_HOURS') TO FIELD-SYMBOL(<weekly_hours>).
      IF sy-subrc EQ 0.
        <weekly_hours> = p0007-wostd.
      ENDIF.

      IF period-endda EQ period-end.
        message_handler = NEW #( ).

        ASSIGN row->('HIRING_DATE') TO FIELD-SYMBOL(<hiring_date>).
        IF sy-subrc EQ 0.
          <hiring_date> = me->get_hiring_date( pernr           = p0001-pernr
                                               message_handler = message_handler ).
        ENDIF.

        ASSIGN row->('ENDING_DATE') TO FIELD-SYMBOL(<ending_date>).
        IF sy-subrc EQ 0.
          <ending_date> = me->get_leaving_date( pernr              = p0001-pernr
                                                selection_end_date = CONV #( period-end + 1 )
                                                message_handler    = message_handler ).
        ENDIF.

        IF me->params-vacation_for_leavers NE abap_true OR <ending_date> IS ASSIGNED AND <ending_date> IS NOT INITIAL.
          ASSIGN row->('VACATION_BALANCE') TO FIELD-SYMBOL(<vacation_balance>).
          IF sy-subrc EQ 0.
            <vacation_balance> = me->calculate_vacation_balance( p2006 ).
          ENDIF.
        ENDIF.

        " Get date for the forced payment from the tech.wage type '9PAY'
        forced_payment_date = me->get_forced_payment_date( b2_cluster ).

        " B2 Cluster data/calculations only for a time frame that contains period end!
        overtime_number = me->get_overtime_number( b2_cluster = b2_cluster
                                                   date       = COND #( WHEN forced_payment_date IS NOT INITIAL
                                                                        THEN forced_payment_date
                                                                        ELSE period-end ) ).

        ASSIGN row->('OVERTIME_NUMBER') TO FIELD-SYMBOL(<overtime_number>).
        IF sy-subrc EQ 0.
          <overtime_number> = overtime_number.
        ENDIF.

        hourly_rate = me->get_hourly_rate( evaluated_wage_types ).

        ASSIGN row->('OVERTIME_PAY') TO FIELD-SYMBOL(<overtime_pay>).
        IF sy-subrc EQ 0.
          <overtime_pay> = me->calculate_overtime_pay( overtime_number = overtime_number
                                                       hourly_rate     = hourly_rate ).
        ENDIF.

        ASSIGN row->('CURRENCY_OVERTIME') TO FIELD-SYMBOL(<currency_overtime>).
        IF sy-subrc EQ 0.
          <currency_overtime> = p0008-waers.
        ENDIF.

        hours_reset_number = me->get_hours_reset_number( b2_cluster = b2_cluster
                                                         date       = period-end ).

        ASSIGN row->('HOURS_RESET') TO FIELD-SYMBOL(<hours_reset>).
        IF sy-subrc EQ 0.
          <hours_reset> = hours_reset_number.
        ENDIF.

        ASSIGN row->('HOURS_COST') TO FIELD-SYMBOL(<hours_cost>).
        IF sy-subrc EQ 0.
          <hours_cost> = me->calculate_hours_cost( hours_reset_number = hours_reset_number
                                                   hourly_rate        = hourly_rate ).
        ENDIF.

        ASSIGN row->('CURRENCY_HOURS') TO FIELD-SYMBOL(<currency_hours>).
        IF sy-subrc EQ 0.
          <currency_hours> = p0008-waers.
        ENDIF.
      ENDIF.

      me->add_row( row ).
    ENDPROVIDE.

  ENDMETHOD.

* <SIGNATURE>---------------------------------------------------------------------------------------+
* | Instance Private Method ZCL_PY_PROVIDER_APP->BUILD_PERIODS
* +-------------------------------------------------------------------------------------------------+
* | [--->] B2_CLUSTER                     TYPE        HRF_TIM_B2
* | [<-()] RESULT                         TYPE        TT_PERIODS
* +--------------------------------------------------------------------------------------</SIGNATURE>
  METHOD build_periods.

    DATA periods TYPE tt_periods.

    " Build unique table of periods (endda = ZL-date)
    periods = VALUE #( FOR GROUPS date OF <zl> IN b2_cluster-ft_zl
                       GROUP BY <zl>-datum WITHOUT MEMBERS ( end = date ) ).

    " Update periods with begin date
    DO lines( periods ) TIMES.
      DATA(index) = sy-index.

      DATA(current)  = REF #( periods[ index ] ).
      DATA(previous) = REF #( periods[ index - 1 ] OPTIONAL ).

      current->begda = COND #( WHEN previous IS NOT BOUND THEN me->params-payroll_begda
                                                          ELSE previous->endda + 1 ).
      current->endda = current->end.
    ENDDO.

    result = periods.

  ENDMETHOD.
